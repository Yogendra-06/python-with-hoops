class Student:
    """
    A comprehensive Student class demonstrating multiple methods and OOP principles.
    """
    
    # Class variable
    total_students = 0
    
    def __init__(self, name, student_id, email, age):
        """
        Constructor method to initialize student attributes.
        
        Args:
            name (str): Student's name
            student_id (int): Unique student ID
            email (str): Student's email
            age (int): Student's age
        """
        self.name = name
        self.student_id = student_id
        self.email = email
        self.age = age
        self.grades = []
        self.courses = []
        Student.total_students += 1
    
    def add_grade(self, subject, grade):
        """
        Add a grade for a specific subject.
        
        Args:
            subject (str): Name of the subject
            grade (float): Grade received (0-100)
        """
        if 0 <= grade <= 100:
            self.grades.append({"subject": subject, "grade": grade})
            print(f"Grade {grade} added for {subject}")
        else:
            print("Invalid grade. Please enter a grade between 0 and 100.")
    
    def enroll_course(self, course_name):
        """
        Enroll student in a course.
        
        Args:
            course_name (str): Name of the course
        """
        if course_name not in self.courses:
            self.courses.append(course_name)
            print(f"Successfully enrolled in {course_name}")
        else:
            print(f"Already enrolled in {course_name}")
    
    def drop_course(self, course_name):
        """
        Drop a course from student's enrollment.
        
        Args:
            course_name (str): Name of the course to drop
        """
        if course_name in self.courses:
            self.courses.remove(course_name)
            print(f"Successfully dropped {course_name}")
        else:
            print(f"Not enrolled in {course_name}")
    
    def calculate_average_grade(self):
        """
        Calculate and return the average grade.
        
        Returns:
            float: Average grade or 0 if no grades
        """
        if not self.grades:
            return 0
        total = sum(grade["grade"] for grade in self.grades)
        return total / len(self.grades)
    
    def get_grade_report(self):
        """
        Get a detailed grade report for all subjects.
        
        Returns:
            dict: Dictionary containing all grades and average
        """
        return {
            "student": self.name,
            "grades": self.grades,
            "average": self.calculate_average_grade()
        }
    
    def display_student_info(self):
        """
        Display complete student information.
        """
        print("\n" + "="*50)
        print(f"Student Information")
        print("="*50)
        print(f"Name: {self.name}")
        print(f"ID: {self.student_id}")
        print(f"Email: {self.email}")
        print(f"Age: {self.age}")
        print(f"Courses Enrolled: {', '.join(self.courses) if self.courses else 'None'}")
        print(f"Average Grade: {self.calculate_average_grade():.2f}")
        print("="*50 + "\n")
    
    def update_email(self, new_email):
        """
        Update student's email address.
        
        Args:
            new_email (str): New email address
        """
        self.email = new_email
        print(f"Email updated to {new_email}")
    
    def update_age(self, new_age):
        """
        Update student's age.
        
        Args:
            new_age (int): New age
        """
        if new_age > 0:
            self.age = new_age
            print(f"Age updated to {new_age}")
        else:
            print("Invalid age. Please enter a positive number.")
    
    def is_passing(self, passing_grade=60):
        """
        Check if student is passing (average grade >= passing grade).
        
        Args:
            passing_grade (int): Minimum grade to pass (default: 60)
            
        Returns:
            bool: True if passing, False otherwise
        """
        average = self.calculate_average_grade()
        return average >= passing_grade
    
    @classmethod
    def get_total_students(cls):
        """
        Class method to get total number of students created.
        
        Returns:
            int: Total number of students
        """
        return cls.total_students
    
    @staticmethod
    def validate_email(email):
        """
        Static method to validate email format.
        
        Args:
            email (str): Email to validate
            
        Returns:
            bool: True if valid email format, False otherwise
        """
        return "@" in email and "." in email
    
    def __str__(self):
        """
        String representation of the student.
        
        Returns:
            str: Student information as string
        """
        return f"Student({self.name}, ID: {self.student_id})"
    
    def __repr__(self):
        """
        Official string representation of the student.
        
        Returns:
            str: Developer-friendly representation
        """
        return f"Student(name='{self.name}', student_id={self.student_id}, email='{self.email}', age={self.age})"
